cmake_minimum_required(VERSION 3.20)
project(CorrectedVE LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
add_executable(corrected_ve cycle_run.cpp)
target_link_libraries(corrected_ve PRIVATE OpenMP::OpenMP_CXX)
target_compile_definitions(corrected_ve PRIVATE OPENWAM_COMMON_ELEMENTAL_EOS)
add_executable(corrected_ve_inspect inspect.cpp)
target_link_libraries(corrected_ve_inspect PRIVATE OpenMP::OpenMP_CXX)
target_compile_definitions(corrected_ve_inspect PRIVATE OPENWAM_COMMON_ELEMENTAL_EOS)
